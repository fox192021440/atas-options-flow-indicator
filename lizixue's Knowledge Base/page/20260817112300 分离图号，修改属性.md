' 所有变量声明
Dim boolstatus As Boolean
Dim longstatus As Long, longwarnings As Long
Dim a As Integer, j As Integer
Dim b As String, m As String, k As String, c As String
Dim strmat As String, tempvalue As String
Dim Part As Object
Dim swApp As Object
Dim swModelDoc As Object
Dim swModelDocExt As Object          ' 必须声明，否则编译错误
Dim swConfig As Object
Dim CustPropMgr As Object
Dim swModel As Object

' 如果需要清空所有属性，保留以下变量
Dim swModel2 As Object
Dim vCustInfoNameArr2 As Variant
Dim vCustInfoName2 As Variant
Dim bRet As Boolean

Sub main()
    Set swApp = Application.SldWorks
    Set swModelDoc = swApp.ActiveDoc
    Set swModelDocExt = swModelDoc.Extension

    ' ----- 可选：清空所有自定义属性（如需启用，请取消注释） -----
    'Set swModel2 = swApp.ActiveDoc
    'vCustInfoNameArr2 = swModel2.GetCustomInfoNames
    'If Not IsEmpty(vCustInfoNameArr2) Then
    '    For Each vCustInfoName2 In vCustInfoNameArr2
    '        bRet = swModel2.DeleteCustomInfo(vCustInfoName2)
    '    Next
    'End If
    ' -----------------------------------------------------------

    Set CustPropMgr = swModelDocExt.CustomPropertyManager("")

    ' 获取当前文件名（含扩展名）
    c = swApp.ActiveDoc.GetTitle()   ' 例如：CS20260812001毛细管.SLDPRT

    ' 用正则提取图号和零件名称
    Dim regex As Object, matches As Object
    Set regex = CreateObject("VBScript.RegExp")
    regex.Pattern = "^([A-Za-z]+\d{8}\d{3})(.+)\.(SLDPRT|SLDASM)$"
    regex.IgnoreCase = True

    If regex.Test(c) Then
        Set matches = regex.Execute(c)
        k = matches(0).SubMatches(0)   ' 图号，如 CS20260812001
        m = matches(0).SubMatches(1)   ' 零件名称，如 毛细管
    Else
        MsgBox "文件名格式不符！期望格式：项目代码+8位日期+3位流水号+零件名称" & vbCrLf & _
               "例如：CS20260812001毛细管.SLDPRT", vbExclamation
        Exit Sub
    End If
    Set regex = Nothing

    ' 删除原有的“图号”“零件名称”属性（如果存在）
    On Error Resume Next
    CustPropMgr.Delete "图号"
    CustPropMgr.Delete "零件名称"
    On Error GoTo 0

    ' 写入新属性（30 = swCustomInfoText）
    CustPropMgr.Add2 "图号", 30, k
    CustPropMgr.Add2 "零件名称", 30, m

    MsgBox "写入成功！" & vbCrLf & "图号：" & k & vbCrLf & "零件名称：" & m, vbInformation, "完成"
End Sub
